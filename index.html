# ============================================================ # HASINEH PRICE ENGINE V1 # FREE MARKET PRICE COLLECTOR # ============================================================ # Project: # HASINEH MARKET # # Purpose: # Collect public market prices # Normalize values # Save clean JSON data # # V1 - Foundation # ============================================================ import json import re import time from datetime import datetime, timezone from urllib.request import Request, urlopen from urllib.error import URLError, HTTPError # ------------------------------------------------------------ # SETTINGS # ------------------------------------------------------------ OUTPUT_FILE = "price-engine/prices.json" REQUEST_TIMEOUT = 20 HEADERS = { "User-Agent": ( "Mozilla/5.0 " "(Windows NT 10.0; Win64; x64) " "AppleWebKit/537.36 " "(KHTML, like Gecko) " "Chrome/120.0 Safari/537.36" ) } # Public market source # We will expand and improve sources in the next engine stages. TGJU_URL = "https://www.tgju.org/" # ------------------------------------------------------------ # PERSIAN / ARABIC DIGITS # ------------------------------------------------------------ PERSIAN_DIGITS = "۰۱۲۳۴۵۶۷۸۹" ARABIC_DIGITS = "٠١٢٣٤٥٦٧٨٩" ENGLISH_DIGITS = "0123456789" def normalize_digits(value): """ Convert Persian and Arabic digits to English digits. """ if value is None: return "" text = str(value) translation_table = str.maketrans( PERSIAN_DIGITS + ARABIC_DIGITS, ENGLISH_DIGITS + ENGLISH_DIGITS ) return text.translate(translation_table) # ------------------------------------------------------------ # NUMBER CLEANING # ------------------------------------------------------------ def clean_number(value): """ Convert a price-like string into an integer. Examples: 123,456,789 ۱۲۳٬۴۵۶٬۷۸۹ 123.456.789 """ if value is None: return None text = normalize_digits(value) text = text.replace(",", "") text = text.replace("٬", "") text = text.replace(" ", "") text = text.replace(".", "") match = re.search(r"\d+", text) if not match: return None try: return int(match.group()) except ValueError: return None # ------------------------------------------------------------ # FETCH PUBLIC PAGE # ------------------------------------------------------------ def fetch_page(url): """ Download a public webpage. Returns: page text or None if the request fails. """ request = Request( url, headers=HEADERS ) try: with urlopen( request, timeout=REQUEST_TIMEOUT ) as response: content = response.read() return content.decode( "utf-8", errors="ignore" ) except HTTPError as error: print( f"[ERROR] HTTP {error.code}: {url}" ) return None except URLError as error: print( f"[ERROR] URL error: {error.reason}" ) return None except Exception as error: print( f"[ERROR] Fetch failed: {error}" ) return None # ------------------------------------------------------------ # SOURCE STATUS # ------------------------------------------------------------ def source_status(page): """ Check whether the source page was successfully received. """ if page is None: return { "online": False, "message": "SOURCE_UNAVAILABLE" } if len(page) < 500: return { "online": False, "message": "SOURCE_RESPONSE_TOO_SHORT" } return { "online": True, "message": "SOURCE_OK" } # ------------------------------------------------------------ # BASIC DATA STRUCTURE # ------------------------------------------------------------ def empty_price_item( name, unit="تومان", category="market" ): """ Standard structure for every price item. """ return { "name": name, "category": category, "price": None, "previous": None, "change": None, "change_percent": None, "unit": unit, "source": None, "status": "pending" } # ------------------------------------------------------------ # INITIAL MARKET ITEMS # ------------------------------------------------------------ def create_market_structure(): items = { "gold_18": empty_price_item( "طلای ۱۸ عیار", "تومان / گرم", "gold" ), "gold_24": empty_price_item( "طلای ۲۴ عیار", "تومان / گرم", "gold" ), "mesghal": empty_price_item( "مثقال طلا", "تومان / مثقال", "gold" ), "coin_emami": empty_price_item( "سکه امامی", "تومان / عدد", "coin" ), "coin_half": empty_price_item( "نیم سکه", "تومان / عدد", "coin" ), "coin_quarter": empty_price_item( "ربع سکه", "تومان / عدد", "coin" ), "silver": empty_price_item( "نقره", "تومان / گرم", "metal" ), "usd": empty_price_item( "دلار آمریکا", "تومان", "currency" ), "eur": empty_price_item( "یورو", "تومان", "currency" ), "aed": empty_price_item( "درهم امارات", "تومان", "currency" ) } return items # ------------------------------------------------------------ # ENGINE METADATA # ------------------------------------------------------------ def engine_metadata(source_info): now = datetime.now( timezone.utc ) return { "engine": "HASINEH PRICE ENGINE", "version": "V1", "mode": "FREE_PUBLIC_SOURCE", "generated_at_utc": now.isoformat(), "source": "TGJU_PUBLIC_PAGE", "source_status": source_info, "currency_display": "IRR_TOMAN", "automatic": True } # ------------------------------------------------------------ # BUILD JSON # ------------------------------------------------------------ def build_output(page): source_info = source_status(page) data = { "meta": engine_metadata( source_info ), "prices": create_market_structure() } # -------------------------------------------------------- # IMPORTANT # -------------------------------------------------------- # At this stage we intentionally DO NOT invent prices. # # The next collector stage will map real source fields # to the items above. # # This prevents fake/demo prices from entering # HASINEH MARKET. # -------------------------------------------------------- if source_info["online"]: for key in data["prices"]: data["prices"][key]["source"] = ( "TGJU_PUBLIC" ) data["prices"][key]["status"] = ( "waiting_for_parser" ) else: for key in data["prices"]: data["prices"][key]["source"] = ( "TGJU_PUBLIC" ) data["prices"][key]["status"] = ( "source_unavailable" ) return data # ------------------------------------------------------------ # SAVE JSON # ------------------------------------------------------------ def save_json(data): with open( OUTPUT_FILE, "w", encoding="utf-8" ) as file: json.dump( data, file, ensure_ascii=False, indent=2 ) # ------------------------------------------------------------ # MAIN ENGINE # ------------------------------------------------------------ def main(): print("") print("=" * 60) print("HASINEH PRICE ENGINE V1") print("FREE MARKET PRICE COLLECTOR") print("=" * 60) print("") print("[1/4] Connecting to public source...") page = fetch_page( TGJU_URL ) if page is not None: print( "[OK] Public source received." ) else: print( "[WARNING] Public source unavailable." ) print( "[2/4] Checking source status..." ) status = source_status( page ) print( "[STATUS]", status["message"] ) print( "[3/4] Building HASINEH price structure..." ) data = build_output( page ) print( "[4/4] Saving prices.json..." ) save_json( data ) print("") print( "[DONE] HASINEH PRICE ENGINE V1 completed." ) print( f"[OUTPUT] {OUTPUT_FILE}" ) print("") print("=" * 60) # ------------------------------------------------------------ # START # ------------------------------------------------------------ if __name__ == "__main__": main()
